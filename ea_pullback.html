MT5 EA Pullback Strategy

#include 
CTrade trade;

// === SETTINGS ===
input double LotSize = 0.01;

input int FastMA = 5;
input int SlowMA = 10;
input int TrendMA = 200;

input int StopLoss = 10;
input int TakeProfit = 25;

input double ProfitTargetUSD = 0.50;
input double MaxLossUSD = 0.30;

input int CooldownSeconds = 180;

// === HANDLES ===
int fastHandle, slowHandle, trendHandle;
double fastBuffer[], slowBuffer[], trendBuffer[];

datetime lastTradeTime = 0;

// === INIT ===
int OnInit()
{
   fastHandle = iMA(_Symbol, _Period, FastMA, 0, MODE_SMA, PRICE_CLOSE);
   slowHandle = iMA(_Symbol, _Period, SlowMA, 0, MODE_SMA, PRICE_CLOSE);
   trendHandle = iMA(_Symbol, _Period, TrendMA, 0, MODE_SMA, PRICE_CLOSE);
   return(INIT_SUCCEEDED);
}

// === MAIN ===
void OnTick()
{
   if(PositionSelect(_Symbol))
   {
      double profit = PositionGetDouble(POSITION_PROFIT);

      if(profit >= ProfitTargetUSD)
      {
         Alert("Trade CLOSED (PROFIT): $", DoubleToString(profit, 2));
         trade.PositionClose(_Symbol);
         return;
      }

      if(profit <= -MaxLossUSD)
      {
         Alert("Trade CLOSED (LOSS): $", DoubleToString(profit, 2));
         trade.PositionClose(_Symbol);
         return;
      }
   }

   if(TimeCurrent() - lastTradeTime < CooldownSeconds)
      return;

   CopyBuffer(fastHandle, 0, 0, 3, fastBuffer);
   CopyBuffer(slowHandle, 0, 0, 3, slowBuffer);
   CopyBuffer(trendHandle, 0, 0, 1, trendBuffer);

   double fastNow = fastBuffer[0];
   double fastPrev = fastBuffer[1];

   double slowNow = slowBuffer[0];
   double slowPrev = slowBuffer[1];

   double trend = trendBuffer[0];

   double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);

   double close = iClose(_Symbol, _Period, 0);
   double prevClose = iClose(_Symbol, _Period, 1);

   if(PositionSelect(_Symbol))
      return;

   bool upTrend = price > trend;
   bool downTrend = price < trend;

   bool pullbackBuy =
      upTrend &&
      fastPrev > slowPrev &&
      fastNow < slowNow &&
      close > prevClose;

   bool pullbackSell =
      downTrend &&
      fastPrev < slowPrev &&
      fastNow > slowNow &&
      close < prevClose;

   if(pullbackBuy)
   {
      double sl = price - StopLoss * _Point;
      double tp = price + TakeProfit * _Point;

      if(trade.Buy(LotSize, _Symbol, 0, sl, tp))
      {
         Alert("BUY OPENED (PULLBACK) at ", DoubleToString(price, _Digits));
         lastTradeTime = TimeCurrent();
      }
   }

   if(pullbackSell)
   {
      double sl = price + StopLoss * _Point;
      double tp = price - TakeProfit * _Point;

      if(trade.Sell(LotSize, _Symbol, 0, sl, tp))
      {
         Alert("SELL OPENED (PULLBACK) at ", DoubleToString(price, _Digits));
         lastTradeTime = TimeCurrent();
      }
   }
}
